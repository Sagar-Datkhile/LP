# RSA Simple Implementation

p = 3
q = 11

n = p * q
phi = (p - 1) * (q - 1)

e = 7

# Find private key d
d = next(i for i in range(phi) if (e * i) % phi == 1)

print("Public Key:", (e, n))
print("Private Key:", (d, n))

# Input must be integer
m = int(input("Enter message (integer < 33): "))

# Encryption
c = pow(m, e, n)
print("Encrypted Message:", c)

# Decryption
m2 = pow(c, d, n)
print("Decrypted Message:", m2)
