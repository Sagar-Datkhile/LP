# -------- BFS --------

from collections import deque

graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F'],
    'D': [],
    'E': ['F'],
    'F': []
}

def bfs(start):
    queue = deque([start])
    visited = []

    while queue:
        node = queue.popleft()

        if node not in visited:
            print(node, end=" ")
            visited.append(node)

            for n in graph[node]:
                queue.append(n)

print("BFS Traversal:")
bfs('A')
