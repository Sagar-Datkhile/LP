Public Module Calculatordll
    Public Function add(ByVal Num1 As Integer, ByVal Num2 As Integer)
        Dim RESULT As Integer
        RESULT = Num1 + Num2
        Return (RESULT)
    End Function
    Public Function Sub1(ByVal Num1 As Integer, ByVal Num2 As Integer)
        Dim RESULT As Integer
        RESULT = Num1 - Num2
        Return (RESULT)
    End Function
    Public Function Mul(ByVal Num1 As Integer, ByVal Num2 As Integer)
        Dim RESULT As Integer
        RESULT = Num1 * Num2
        Return (RESULT)
    End Function
    Public Function Div(ByVal Num1 As Integer, ByVal Num2 As Integer)
        Dim RESULT As Integer
        RESULT = Num1 / Num2
        Return (RESULT)
    End Function
End Module
